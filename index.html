  Title: string;
  Department: string;
}

const EmployeeDirectory: React.FC = () => {
  const [employees, setEmployees] =
    React.useState<IEmployee[]>([]);

  React.useEffect(() => {
    sp.web.lists
      .getByTitle('Employees')
      .items.select('Title','Department')()
      .then(setEmployees);
  }, []);

  return (
    <ul>
      {employees.map(e => (
        <li key={e.Id}>{e.Title}</li>
      ))}
    </ul>
  );
};
